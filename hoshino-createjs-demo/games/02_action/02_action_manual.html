1. ゲーム概要と実行画面

  • ゲーム内容: 自機（青い四角形）を操作し、障害物（赤のトゲ）や落とし穴を避けながらコインを獲得し、右上のゴール旗を目指す2D物理アクションゲームです。
  • 操作方法: ← / → キー（左右移動）、↑ / W / スペース（ジャンプ）。
  • 物理ルール: 重力 (GRAVITY = 0.6)、ジャンプ初速 (JUMP_POWER = -12)、接地判定。

const STAGE_W = 960;
const STAGE_H = 540;

const PLAYER_SPEED = 6;
const JUMP_POWER = -12;
const GRAVITY = 0.6;
const PLAYER_SIZE = 30;

let stage;
let scene_id = 0;
let score = 0;

let player;
let playerVX = 0;
let playerVY = 0;
let isGrounded = false;

let platforms = [];
let coins = [];
let hazards = [];
let goal;

let isPressLeft = false;
let isPressRight = false;
let isPressJump = false;
let scoreText;

行別詳細解説

  • `PLAYER_SPEED = 6` (移動速度), `JUMP_POWER = -12` (ジャンプ初速), `GRAVITY = 0.6` (重力加速度) で物理挙動をパラメーター管理します。
  • `playerVX` / `playerVY` で速度ベクトル、`isGrounded` で接地フラグを保持します。

3.3 init.js

window.addEventListener("load", init);

function init() {
    stage = new createjs.Stage("myCanvas");
    window.addEventListener("keydown", handleKeyDown);
    window.addEventListener("keyup", handleKeyUp);
    initTitle();
    createjs.Ticker.timingMode = createjs.Ticker.RAF;
    createjs.Ticker.addEventListener("tick", handleTick);
}

function initTitle() {
    scene_id = 0;
    stage.removeAllChildren();
    let bg = new createjs.Shape();
    bg.graphics.beginFill("#e6f7ff").drawRect(0, 0, STAGE_W, STAGE_H);
    stage.addChild(bg);
    let titleText = new createjs.Text("アクションゲーム", "40px bold sans-serif", "blue");
    titleText.x = STAGE_W / 2 - titleText.getMeasuredWidth() / 2;
    titleText.y = 150;
    stage.addChild(titleText);
    stage.addEventListener("stagemousedown", startGame);
    stage.update();
}

function startGame() {
    if (scene_id !== 0) return;
    scene_id = 1;
    stage.removeAllChildren();
    stage.removeEventListener("stagemousedown", startGame);
    score = 0;
    platforms = []; coins = []; hazards = [];
    createStageLayout();
    createPlayer();
    scoreText = new createjs.Text("スコア: 0", "26px bold sans-serif", "black");
    scoreText.x = 20; scoreText.y = 20;
    stage.addChild(scoreText);
}

function createPlayer() {
    player = new createjs.Shape();
    player.graphics.beginFill("blue").drawRect(-PLAYER_SIZE/2, -PLAYER_SIZE/2, PLAYER_SIZE, PLAYER_SIZE);
    player.x = 60; player.y = 400;
    playerVX = 0; playerVY = 0; isGrounded = false;
    stage.addChild(player);
}

function createStageLayout() {
    let platformData = [
        { x: 0, y: 460, w: 960, h: 80 },
        { x: 160, y: 370, w: 140, h: 20 },
        { x: 370, y: 290, w: 160, h: 20 },
        { x: 600, y: 210, w: 150, h: 20 },
        { x: 800, y: 150, w: 130, h: 20 }
    ];
    for (let data of platformData) {
        let p = new createjs.Shape();
        p.graphics.beginFill("green").drawRect(0, 0, data.w, data.h);
        p.x = data.x; p.y = data.y; p.w = data.w; p.h = data.h;
        stage.addChild(p); platforms.push(p);
    }
}

行別詳細解説

  • `createPlayer()` で自機四角形を作成し `(60, 400)` に初期配置します。
  • `createStageLayout()` で配列定義された座標データをループ生成し、足場・トゲ・コイン・ゴール旗を構築します。

3.4 tick.js

function handleTick(event) {
    if (scene_id === 1) {
        playerVX = 0;
        if (isPressLeft) playerVX = -PLAYER_SPEED;
        if (isPressRight) playerVX = PLAYER_SPEED;

        if (isPressJump && isGrounded) {
            playerVY = JUMP_POWER;
            isGrounded = false;
        }

        playerVY += GRAVITY;
        player.x += playerVX;
        player.y += playerVY;

        if (player.x - PLAYER_SIZE/2 < 0) player.x = PLAYER_SIZE/2;
        if (player.x + PLAYER_SIZE/2 > STAGE_W) player.x = STAGE_W - PLAYER_SIZE/2;

        checkPlatformCollision();
        if (player.y > STAGE_H + 50) {
            showGameOver(false);
            return;
        }
        checkHazardCollision();
        checkCoinCollision();
        checkGoalCollision();
    }
    stage.update();
}

function checkPlatformCollision() {
    isGrounded = false;
    for (let p of platforms) {
        let feetY = player.y + PLAYER_SIZE / 2;
        let prevFeetY = feetY - playerVY;
        if (player.x + PLAYER_SIZE/3 >= p.x && player.x - PLAYER_SIZE/3 <= p.x + p.w) {
            if (prevFeetY <= p.y + 6 && feetY >= p.y && playerVY >= 0) {
                player.y = p.y - PLAYER_SIZE / 2;
                playerVY = 0;
                isGrounded = true;
                break;
            }
        }
    }
}

行別詳細解説

  • `playerVY += GRAVITY` で重力落下計算。
  • `checkPlatformCollision()` では前フレームの足元 `prevFeetY` が足場上面 `p.y` より上であり、現在 `feetY` が足場上面に達した時のみ着地 (`isGrounded = true`) と判定し、すり抜けを防ぎます。

3.5 key.js

function handleKeyDown(event) {
    if (event.keyCode === 37 || event.keyCode === 65) isPressLeft = true;
    if (event.keyCode === 39 || event.keyCode === 68) isPressRight = true;
    if (event.keyCode === 38 || event.keyCode === 87 || event.keyCode === 32) {
        isPressJump = true;
        if (scene_id === 0) startGame();
        if (scene_id >= 2) initTitle();
    }
}
function handleKeyUp(event) {
    if (event.keyCode === 37 || event.keyCode === 65) isPressLeft = false;
    if (event.keyCode === 39 || event.keyCode === 68) isPressRight = false;
    if (event.keyCode === 38 || event.keyCode === 87 || event.keyCode === 32) isPressJump = false;
}

行別詳細解説

  • `keyCode` (37:左, 39:右, 38/32:ジャンプ) の状態フラグ操作およびタイトル・結果画面からの画面遷移を処理します。

3.6 gameover.js

function showGameOver(isClear) {
    scene_id = isClear ? 3 : 2;
    if (isClear) playSE("../../Audio/ラッパのファンファーレ.mp3");

    let msg = isClear ? "STAGE CLEAR!" : "GAME OVER";
    let color = isClear ? "green" : "red";

    let resultText = new createjs.Text(msg, "44px bold sans-serif", color);
    resultText.x = STAGE_W / 2 - resultText.getMeasuredWidth() / 2;
    resultText.y = 180;
    stage.addChild(resultText);

    setTimeout(function() {
        stage.addEventListener("stagemousedown", initTitle);
    }, 500);
}

行別詳細解説

  • クリア時 (`isClear === true`) にファンファーレを鳴らし `STAGE CLEAR!` を表示、失敗時は `GAME OVER` を表示して、500msの遅延後にタイトルに戻れるようにします。
